static double dominantPeriod(const std::vector<float>& x, int start, int minLag, int maxLag)
{
    // нормализованная автокорреляция
    double best = 0; int bestLag = 0;
    for (int lag = minLag; lag <= maxLag; ++lag) {
        double num = 0, d1 = 0, d2 = 0;
        for (int i = start; i + lag < (int) x.size(); i += 3) { num += x[i]*x[i+lag]; d1 += x[i]*x[i]; d2 += x[i+lag]*x[i+lag]; }
        const double r = num / std::sqrt(d1*d2 + 1e-12);
        if (r > best) { best = r; bestLag = lag; }
    }
    return bestLag / SR * 1000.0;  // ms
}
