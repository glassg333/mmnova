cmake_minimum_required(VERSION 3.22)
project(MonoJuice VERSION 0.8.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(NOVA_BUILD_PLUGIN "Build JUCE plugin" ON)
option(NOVA_BUILD_TESTS "Build tests" ON)
option(NOVA_SYSTEM_CODECS "Link external FLAC/Ogg/Vorbis for system-packaged JUCE on Linux" OFF)
option(NOVA_LOW_MEMORY_BUILD "Compile JUCE GUI basics with -O1 on non-MSVC builds" OFF)
add_library(NovaDSP STATIC Source/dsp/ChorusCore.cpp Source/dsp/AudioAdapter.cpp)
target_include_directories(NovaDSP PUBLIC Source)
if(MSVC)
 target_compile_options(NovaDSP PRIVATE /fp:precise)
else()
 target_compile_options(NovaDSP PRIVATE -fno-fast-math -ffp-contract=off)
endif()
if(NOVA_BUILD_TESTS)
 enable_testing()
 add_executable(ChorusTests tests/ChorusTests.cpp)
 target_link_libraries(ChorusTests PRIVATE NovaDSP)
 add_test(NAME OriginalChorusGolden COMMAND ChorusTests)
 add_executable(FXTests tests/FXTests.cpp)
 target_link_libraries(FXTests PRIVATE NovaDSP)
 add_test(NAME FXChain COMMAND FXTests)
 add_executable(RackTests tests/RackTests.cpp)
 add_library(ParallelCombOracle STATIC tests/CombOracle.cpp Reference/Uhhyou/ParallelComb/source/dsp/dspcore.cpp Reference/Uhhyou/ParallelComb/source/parameter.cpp)
 set_target_properties(ParallelCombOracle PROPERTIES CXX_STANDARD 20)
 target_compile_definitions(ParallelCombOracle PRIVATE TEST_DSP)
 target_link_libraries(RackTests PRIVATE NovaDSP ParallelCombOracle)
 add_test(NAME FX5Rack COMMAND RackTests)
 add_executable(FactoryTests tests/FactoryTests.cpp)
 target_link_libraries(FactoryTests PRIVATE NovaDSP)
 add_test(NAME MonoJuiceFactory COMMAND FactoryTests)
endif()
if(NOVA_BUILD_PLUGIN)
 if(EXISTS "${JUCE_PATH}/CMakeLists.txt")
  add_subdirectory("${JUCE_PATH}" "${CMAKE_BINARY_DIR}/juce" EXCLUDE_FROM_ALL)
 else()
  find_package(JUCE CONFIG REQUIRED)
 endif()
 juce_add_plugin(MonoJuice
  COMPANY_NAME "MonomachineNova" BUNDLE_ID "com.monomachinenova.fx5"
  PLUGIN_MANUFACTURER_CODE MnOn PLUGIN_CODE NvF5
  PRODUCT_NAME "MonoJuice"
  IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
  VST3_CATEGORIES Fx Modulation PitchShift FORMATS VST3 Standalone COPY_PLUGIN_AFTER_BUILD FALSE)
 if(NOVA_LOW_MEMORY_BUILD AND NOT MSVC)
  set_source_files_properties("${JUCE_MODULES_DIR}/juce_gui_basics/juce_gui_basics.cpp" PROPERTIES COMPILE_OPTIONS "-O1")
 endif()
 juce_generate_juce_header(MonoJuice)
 target_sources(MonoJuice PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
 target_compile_definitions(MonoJuice PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
 target_link_libraries(MonoJuice PRIVATE NovaDSP juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
 if(NOVA_SYSTEM_CODECS AND UNIX)
  target_link_libraries(MonoJuice PRIVATE FLAC vorbisenc vorbisfile vorbis ogg)
 endif()
 if(NOVA_BUILD_TESTS)
  add_executable(WrapperTests tests/WrapperTests.cpp)
  target_include_directories(WrapperTests PRIVATE Source "${JUCE_MODULES_DIR}" "${CMAKE_BINARY_DIR}/MonoJuice_artefacts/JuceLibraryCode")
  target_compile_definitions(WrapperTests PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
  target_link_libraries(WrapperTests PRIVATE MonoJuice NovaDSP)
  find_program(NOVA_XVFB_RUN xvfb-run)
  if(UNIX AND NOT APPLE AND NOVA_XVFB_RUN)
   add_test(NAME JUCEWrapper COMMAND "${NOVA_XVFB_RUN}" -a $<TARGET_FILE:WrapperTests> "${CMAKE_BINARY_DIR}/screenshots")
  else()
   add_test(NAME JUCEWrapper COMMAND WrapperTests "${CMAKE_BINARY_DIR}/screenshots")
  endif()
 endif()
endif()
