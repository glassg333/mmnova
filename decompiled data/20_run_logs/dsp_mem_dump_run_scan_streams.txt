#!/usr/bin/env python3
"""Брутфорс-скан прошивки: найти ВСЕ позиции, откуда депакер даёт >N байт без ошибки.
Каждый кандидат запускается с капом на выход; кандидаты сгущаются (перекрытия выкидываются)."""
import sys
sys.path.insert(0, "/home/z/my-project/scripts")
from mm_depack_fast import depack

BIN = "/home/z/my-project/hi08/firmware.bin"
data = open(BIN, "rb").read()
N = len(data)

MIN_OUT = 4000          # минимальный размер, чтобы считать поток реальным
CAP = 600_000

def try_depack(ip, cap=CAP):
    """Депак с ранним абортом по капу через укороченный буфер."""
    out = bytearray()
    tag = 0
    last_dist = 1
    lb = data
    try:
        while True:
            tag = (tag << 1) & 0xFFFFFFFF
            if (tag & 0xFF) == 0:
                if ip >= N: return None
                b = lb[ip]; ip += 1
                tag = ((b << 1) | 1) & 0xFFFFFFFF
                bit = (b >> 7) & 1
            else:
                bit = (tag >> 8) & 1
            if bit:
                if ip >= N: return None
                out.append(lb[ip]); ip += 1
                if len(out) > cap: return None
                continue
            v = 1
            while True:
                tag = (tag << 1) & 0xFFFFFFFF
                if (tag & 0xFF) == 0:
                    if ip >= N: return None
                    b = lb[ip]; ip += 1
                    tag = ((b << 1) | 1) & 0xFFFFFFFF
                    v = v * 2 + (b >> 7)
                else:
                    v = v * 2 + ((tag >> 8) & 1)
                tag = (tag << 1) & 0xFFFFFFFF
                if (tag & 0xFF) == 0:
                    if ip >= N: return None
                    b = lb[ip]; ip += 1
                    tag = ((b << 1) | 1) & 0xFFFFFFFF
                    stop = (b >> 7) & 1
                else:
                    stop = (tag >> 8) & 1
                if stop:
                    break
            if v == 2:
                dist = last_dist
            else:
                if ip >= N: return None
                b = lb[ip]
                off = (v << 8) + b - 768
                if off == 0xFF:
                    return (bytes(out), ip)     # EOF — нормальное завершение
                ip += 1
                dist = off + 1
                last_dist = dist
                if dist > len(out) + 4_000_000: return None
            bits = 0
            for _ in range(2):
                tag = (tag << 1) & 0xFFFFFFFF
                if (tag & 0xFF) == 0:
                    if ip >= N: return None
                    b = lb[ip]; ip += 1
                    tag = ((b << 1) | 1) & 0xFFFFFFFF
                    bits = (bits << 1) | (b >> 7)
                else:
                    bits = (bits << 1) | ((tag >> 8) & 1)
            if bits:
                L = bits
            else:
                v = 1
                while True:
                    tag = (tag << 1) & 0xFFFFFFFF
                    if (tag & 0xFF) == 0:
                        if ip >= N: return None
                        b = lb[ip]; ip += 1
                        tag = ((b << 1) | 1) & 0xFFFFFFFF
                        v = v * 2 + (b >> 7)
                    else:
                        v = v * 2 + ((tag >> 8) & 1)
                    tag = (tag << 1) & 0xFFFFFFFF
                    if (tag & 0xFF) == 0:
                        if ip >= N: return None
                        b = lb[ip]; ip += 1
                        tag = ((b << 1) | 1) & 0xFFFFFFFF
                        stop = (b >> 7) & 1
                    else:
                        stop = (tag >> 8) & 1
                    if stop:
                        break
                L = v + 2
            if dist > 3328:
                L += 1
            n = L + 1
            if n > cap: return None
            cp = len(out) - dist
            if cp < 0: return None
            if cp + n <= len(out):
                out += out[cp:cp + n]
            else:
                block = bytes(out[cp:])
                q, r = divmod(n, len(block))
                out += block * q + block[:r]
            if len(out) > cap: return None
    except (IndexError, ValueError):
        return None

if __name__ == "__main__":
    hits = []
    # сканируем каждый байт — дорого, но с ранним абортом приемлемо; ограничим область 0x4000..0x120000
    import time
    t0 = time.perf_counter()
    for ip in range(0x4000, min(N, 0x120000)):
        r = try_depack(ip)
        if r and len(r[0]) >= MIN_OUT:
            hits.append((ip, len(r[0]), r[1]))
            print(f"ПОТОК @bits={ip:#x} -> {len(r[0])} байт (eof@{r[1]:#x})", flush=True)
        if ip % 0x10000 == 0:
            print(f"  ...{ip:#x} ({time.perf_counter()-t0:.0f}s)", flush=True)
    print(f"готово за {time.perf_counter()-t0:.0f}s, найдено {len(hits)}")
