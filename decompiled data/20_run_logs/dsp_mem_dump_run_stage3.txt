#!/usr/bin/env python3
"""Итерация 4, этап 3: blk1/blk2 записи, нулевость непокрытого, diff DSP1/DSP2, blk4."""
import sys, json, hashlib
import numpy as np
sys.path.insert(0, "/home/z/my-project/scripts")
from mm_iter4_pipeline import words_le24, words_be24, BANK, parse_records

OUT = "/home/z/my-project/download/hi08_iteration4"
HI = "/home/z/my-project/hi08"
streams = {k: open(f"{OUT}/blk{k}_full.bin", "rb").read() for k in range(5)}

# ---------- 1) все записи blk1/blk2
for k in (1, 2):
    recs, consumed, tail = parse_records(streams[k])
    print(f"--- blk{k}: {len(recs)} записей, consumed {consumed}/{len(streams[k])}, tail={tail}")
    for r in recs:
        if r["cmd"] in (3, 4):
            print(f"   cmd{r['cmd']} addr=${r['addr']:06x}")
        else:
            print(f"   {BANK[r['cmd']]}:${r['addr']:06x} +{r['count']:<6} (off {r['off']*3})")

# ---------- 2) непокрытые области образов — нули?
p1 = np.frombuffer(open(f"{HI}/dsp1_pmem.bin", "rb").read(), dtype=np.uint8)
w1 = (p1.reshape(-1,3).astype(np.uint32)[:,0]<<16) | (p1.reshape(-1,3).astype(np.uint32)[:,1]<<8) | p1.reshape(-1,3).astype(np.uint32)[:,2]
p2 = np.frombuffer(open(f"{HI}/dsp2_pmem.bin", "rb").read(), dtype=np.uint8)
img2 = p2.reshape(-1,3).astype(np.uint32)
w2 = (img2[:,0]<<16) | (img2[:,1]<<8) | img2[:,2]

covered = np.zeros(len(w1), bool)
recs3, _, _ = parse_records(streams[3])
for r in recs3:
    if r["cmd"] == 0 and r["count"]:
        covered[r["addr"]:r["addr"]+r["count"]] = True
# blk1/blk2 внутренние P
for k in (1, 2):
    recs, _, _ = parse_records(streams[k])
    for r in recs:
        if r["cmd"] == 0 and r["count"]:
            covered[r["addr"]:r["addr"]+r["count"]] = True

unc = np.where(~covered)[0]
nz_unc = unc[w1[unc] != 0]
print(f"\nнепокрыто {len(unc)} слов, из них НЕнулевых в dsp1: {len(nz_unc)}")
if len(nz_unc):
    runs = []
    s = p = nz_unc[0]
    for x in nz_unc[1:]:
        if x <= p + 4: p = x
        else: runs.append((s,p)); s=p=x
    runs.append((s,p))
    for a,b in runs[:30]:
        print(f"  dsp1 непокрытое НЕнулевое: ${a:06x}..${b:06x}")
nz_unc2 = unc[w2[unc] != 0]
print(f"непокрытых НЕнулевых в dsp2: {len(nz_unc2)}")

# ---------- 3) diff DSP1 vs DSP2 (все слова)
d = np.where(w1 != w2)[0]
print(f"\ndiff dsp1 vs dsp2: {len(d)} слов")
if len(d):
    runs = []
    s = p = d[0]
    for x in d[1:]:
        if x == p+1: p = x
        else: runs.append((s,p)); s=p=x
    runs.append((s,p))
    for a,b in runs[:40]:
        print(f"  ${a:06x}..${b:06x} ({b-a+1} слов)")
        print(f"    dsp1: {' '.join(f'{x:06x}' for x in w1[a:min(a+4,b+1)])}")
        print(f"    dsp2: {' '.join(f'{x:06x}' for x in w2[a:min(a+4,b+1)])}")

# ---------- 4) blk4 структура
b4 = streams[4]
print(f"\n--- blk4: {len(b4)} байт ({len(b4)/3:.1f} слов при /3)")
w4 = words_be24(b4[:len(b4)//3*3])
print(f"первые 24 слова BE24: {' '.join(f'{x:06x}' for x in w4[:24])}")
nz = np.where(w4 != 0)[0]
print(f"ненулевых слов: {len(nz)}/{len(w4)}")
# крупные нулевые и ненулевые прогоны
mask = w4 != 0
edges = np.diff(np.concatenate([[0], mask.view(np.int8), [0]]))
starts = np.where(edges == 1)[0]; ends = np.where(edges == -1)[0]
runs = sorted(((int(s), int(e-s)) for s, e in zip(starts, ends)), key=lambda t: -t[1])
print("топ-15 ненулевых прогонов (адрес_слова, длина):")
for s, l in runs[:15]:
    print(f"  word {s:7d} (${s:06x}) длина {l}")
# строки?
import re
strs = re.findall(rb"[ -~]{6,}", b4)
print(f"строк в blk4: {len(strs)}; первые 30:")
for s in strs[:30]:
    print("   ", s[:60])
