#!/usr/bin/env python3
"""Итерация 4, этап 4: извлечение новых данных.
1) Банк заводских вейвформ (BIN 0x100000..0x15FEFE) — 63 именованные таблицы.
2) Полная карта записей blk1/blk2/blk3 (loader map).
3) Diff ядра DSP1 vs DSP2.
4) Банк заводских пресетов blk4 — разметка.
5) Math-таблицы (sine 8K, width chorus B) — контрольные генерации.
"""
import sys, os, json, re, hashlib
import numpy as np
sys.path.insert(0, "/home/z/my-project/scripts")
from mm_iter4_pipeline import words_le24, words_be24, parse_records, BANK

OUT = "/home/z/my-project/download/hi08_iteration4"
HI = "/home/z/my-project/hi08"
WAV = os.path.join(OUT, "factory_wavetables"); os.makedirs(WAV, exist_ok=True)
REC = os.path.join(OUT, "loader_records"); os.makedirs(REC, exist_ok=True)

data = open(f"{HI}/firmware.bin", "rb").read()
blob = data[0x100000:0x15feff]

# ---------- 1) вейвформы: имена = 4+ символов A-Z0-9, конец = начало следующего
names = []
for m in re.finditer(rb"[A-Z][A-Z0-9]{3}[A-Z0-9 \-]*", blob):
    s = m.group()
    # обрезаем хвостовые не-именные символы (данные могут выглядеть как ASCII)
    core = re.match(rb"[A-Z][A-Z0-9]{3,7}", s)
    if core:
        names.append([m.start(), core.group().decode()])
# дедупликация/слияние: имена должны идти с шагом ~0x17F8; выкидываем ложные
clean = [names[0]]
for off, nm in names[1:]:
    if off - clean[-1][0] >= 0x1000:
        clean.append([off, nm])
print(f"найдено вейвформ: {len(clean)}")
manifest = []
for i, (off, nm) in enumerate(clean):
    end = clean[i + 1][0] if i + 1 < len(clean) else len(blob)
    chunk = blob[off + len(nm):end]
    # имя+заголовок: для первого входа 4 байта до имени
    hdr = blob[off-4:off] if i == 0 else b""
    base = os.path.join(WAV, f"wt{i:02d}_{nm}")
    open(base + ".bin", "wb").write(chunk)
    w = words_be24(chunk[:len(chunk)//3*3])
    txt = [f"# Factory wavetable '{nm}'",
           f"# BIN offset: {0x100000+off:#x}..{0x100000+end:#x} ({len(chunk)} байт)",
           f"# words (BE24): {len(w)}, nonzero: {int((w!=0).sum())}",
           f"# SHA-256: {hashlib.sha256(chunk).hexdigest()}",
           ""]
    for j in range(0, len(w), 8):
        txt.append(f"  +{j:04x}: " + " ".join(f"{x:06x}" for x in w[j:j+8]))
    open(base + ".txt", "w").write("\n".join(txt))
    manifest.append({"index": i, "name": nm, "bin_off": 0x100000 + off,
                     "size": len(chunk), "sha256": hashlib.sha256(chunk).hexdigest()})
    print(f"  #{i:02d} {nm:<8} @{0x100000+off:#x} {len(chunk)} байт")
json.dump(manifest, open(os.path.join(WAV, "_wavetables_manifest.json"), "w"), indent=1)

# ---------- 2) полная карта записей
streams = {k: open(f"{OUT}/blk{k}_full.bin", "rb").read() for k in (1, 2, 3)}
BLK_NAMES = {1: "DSP1_internal", 2: "DSP2_internal", 3: "DSP_common_external"}
full_map = {}
for k in (1, 2, 3):
    recs, consumed, tail = parse_records(streams[k])
    assert tail == 0, f"blk{k} не дочитан: {tail}"
    full_map[BLK_NAMES[k]] = recs
    lines = [f"# {BLK_NAMES[k]}: {len(recs)} записей, поток {len(streams[k])} байт (100% разобран)", ""]
    words_pos = 0
    for r in recs:
        if r["cmd"] == 3:
            lines.append(f"JMP P:${r['addr']:06x}")
        elif r["cmd"] == 4:
            lines.append("END")
        else:
            lines.append(f"{BANK[r['cmd']]} ${r['addr']:06x}  +{r['count']:<6} слов  "
                         f"[{r['off']*3}..{r['off']*3+9+r['count']*3}) sha256={r['sha256'][:12]}")
    open(f"{REC}/{BLK_NAMES[k]}.txt", "w").write("\n".join(lines))
    print(f"{BLK_NAMES[k]}: {len(recs)} записей -> loader_records/")
json.dump(full_map, open(f"{REC}/_records_full.json", "w"), indent=1)

# ---------- 3) diff ядра DSP1 vs DSP2
p1 = np.frombuffer(open(f"{HI}/dsp1_pmem.bin","rb").read(), dtype=np.uint8).reshape(-1,3).astype(np.uint32)
p2 = np.frombuffer(open(f"{HI}/dsp2_pmem.bin","rb").read(), dtype=np.uint8).reshape(-1,3).astype(np.uint32)
w1 = (p1[:,0]<<16)|(p1[:,1]<<8)|p1[:,2]
w2 = (p2[:,0]<<16)|(p2[:,1]<<8)|p2[:,2]
d = np.where(w1 != w2)[0]
difflines = [f"# Diff ядра DSP1 vs DSP2 (P-банк): {len(d)} различающихся слов", ""]
i = 0
while i < len(d):
    j = i
    while j+1 < len(d) and d[j+1] == d[j]+1:
        j += 1
    a, b = d[i], d[j]
    difflines.append(f"P:${a:06x}..${b:06x} ({b-a+1} слов):")
    difflines.append(f"  DSP1: " + " ".join(f"{x:06x}" for x in w1[a:b+1]))
    difflines.append(f"  DSP2: " + " ".join(f"{x:06x}" for x in w2[a:b+1]))
    i = j + 1
open(f"{OUT}/dsp1_vs_dsp2_kernel_diff.txt", "w").write("\n".join(difflines))
print(f"diff DSP1/DSP2: {len(d)} слов -> dsp1_vs_dsp2_kernel_diff.txt")

# ---------- 4) math-таблицы (контрольные)

    arr = np.asarray(arr, dtype=np.int64) & 0xFFFFFF
    b = arr.astype(np.uint32).to_bytes(arr.dtype.itemsize*0+0, 'little') if False else None
    out = bytearray()
    for x in arr:
        out += int(x).to_bytes(3, 'big')
    open(path, 'wb').write(bytes(out))

N = 8192
sine = np.round(np.sin(2*np.pi*np.arange(N)/N) * 0x7FFFFF).astype(np.int64)
cosw = np.round(np.cos(2*np.pi*np.arange(N)/N) * 0x7FFFFF).astype(np.int64)
save24(os.path.join(OUT, "X_14A000_sine_table_8K_math.bin"), sine)
save24(os.path.join(OUT, "X_14A800_width_table_chorus_B_math.bin"), cosw)
print("math-таблицы sine/cos 8K сгенерированы")
print("ГОТОВО")
