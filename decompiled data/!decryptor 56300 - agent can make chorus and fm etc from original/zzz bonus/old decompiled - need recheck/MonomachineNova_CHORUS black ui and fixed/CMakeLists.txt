cmake_minimum_required(VERSION 3.22)
project(MonomachineNovaChorus VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(NOVA_BUILD_PLUGIN "Build the JUCE VST3 and Standalone" ON)
option(NOVA_BUILD_TESTS "Build native tests" ON)
add_library(ChorusDSP STATIC Source/dsp/ChorusCore.cpp Source/dsp/AudioAdapter.cpp)
target_include_directories(ChorusDSP PUBLIC Source)
if(MSVC)
 target_compile_options(ChorusDSP PRIVATE /fp:precise)
else()
 target_compile_options(ChorusDSP PRIVATE -fno-fast-math -ffp-contract=off)
endif()
if(NOVA_BUILD_TESTS)
 enable_testing()
 add_executable(ChorusTests tests/ChorusTests.cpp)
 target_link_libraries(ChorusTests PRIVATE ChorusDSP)
 add_test(NAME ChorusNative COMMAND ChorusTests)
endif()
if(NOVA_BUILD_PLUGIN)
 if(NOT EXISTS "${JUCE_PATH}/CMakeLists.txt")
  message(FATAL_ERROR "Set JUCE_PATH to JUCE 7/8. Projucer/Visual Studio users should open MonomachineNova.jucer instead.")
 endif()
 if(EXISTS "${NOVA_PREBUILT_JUCEAIDE}")
  # Optional local build acceleration. Normal users need only JUCE_PATH.
  set(JUCE_MODULES_ONLY ON CACHE BOOL "" FORCE)
  add_subdirectory("${JUCE_PATH}" "${CMAKE_BINARY_DIR}/juce" EXCLUDE_FROM_ALL)
  include("${JUCE_PATH}/extras/Build/CMake/JUCEUtils.cmake")
  add_executable(juce::juceaide IMPORTED GLOBAL)
  set_target_properties(juce::juceaide PROPERTIES IMPORTED_LOCATION "${NOVA_PREBUILT_JUCEAIDE}")
 else()
  set(JUCE_MODULES_ONLY OFF CACHE BOOL "" FORCE)
  add_subdirectory("${JUCE_PATH}" "${CMAKE_BINARY_DIR}/juce" EXCLUDE_FROM_ALL)
 endif()
 juce_add_plugin(MonomachineNovaChorus
  COMPANY_NAME "MonomachineNova" BUNDLE_ID "com.monomachinenova.chorus"
  PLUGIN_MANUFACTURER_CODE MnOn PLUGIN_CODE NvCh
  PRODUCT_NAME "MonomachineNova CHORUS"
  IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
  VST3_CATEGORIES Fx Modulation FORMATS VST3 Standalone COPY_PLUGIN_AFTER_BUILD FALSE)
 juce_generate_juce_header(MonomachineNovaChorus)
 target_sources(MonomachineNovaChorus PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
 target_compile_definitions(MonomachineNovaChorus PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
 target_link_libraries(MonomachineNovaChorus PRIVATE ChorusDSP juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
 if(NOVA_BUILD_TESTS)
  add_executable(ChorusWrapperTests tests/WrapperTests.cpp)
  target_include_directories(ChorusWrapperTests PRIVATE Source "${JUCE_PATH}/modules" "${CMAKE_BINARY_DIR}/MonomachineNovaChorus_artefacts/JuceLibraryCode")
  target_compile_definitions(ChorusWrapperTests PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
  target_link_libraries(ChorusWrapperTests PRIVATE MonomachineNovaChorus ChorusDSP)
 endif()
endif()
