# ============================================================================
#  Подключение модуля mmnova к Monomachine-Nova-1.6.0

target_sources(MonomachineNova PRIVATE
    Source/mmnova/mmnova.cpp)          # unity build, остальное header-only

target_include_directories(MonomachineNova PRIVATE
    Source)

# Если вы выгрузили таблицы прошивки:
#   python3 Source/mmnova/tools/extract_tables_fixed.py \
#       --pmem "<путь>/dsp1_pmem.bin" --out Source/mmnova/MnMFirmwareTables.h
# то включите точный путь:
# target_compile_definitions(MonomachineNova PRIVATE MMNOVA_HAVE_FW_TABLES=1)

# Только нативные 44.1 кГц (без интерполяции ресемплера):
# target_compile_definitions(MonomachineNova PRIVATE MMNOVA_NATIVE_RATE_ONLY=1)

target_compile_features(MonomachineNova PRIVATE cxx_std_17)

if (MSVC)
    target_compile_options(MonomachineNova PRIVATE /fp:fast)
else()
    target_compile_options(MonomachineNova PRIVATE -ffast-math -fno-finite-math-only)
endif()
